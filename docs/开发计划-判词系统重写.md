### 3.3 新函数签名

```python
# ---- 产品评判 ----
def _product_signals(mapped) -> dict[str, Any]
    """提取 6 维信号 + 每维标签。返回：
    {sold, repurchase, price, moq, positive, wanted, return7day, has_service_labels,
     d1_score, d2_score, d3_score, d4_score, d5_score, d6_score,
     d1_label, d2_label, d3_label, d4_label, d5_label, d6_label}
    """

def _product_fatal(signals) -> dict[str, Any] | None
    """致命短板检查。
    - return7day ≠ OK ∧ has_service_labels → 致命（不支持退货）
    - positive_rate < 80% → 致命（差评过多，接口保留）
    命中返回 fatal verdict dict，否则 None。
    """

def _product_missing(signals) -> bool
    """数据不足检查：sold + repurchase + return7day ≥ 2 项缺失 → True"""

def _product_tier(signals) -> str
    """12 条规则顺序匹配 → 返回 tier key。
    规则（按优先级）：
      Rule 3:  repurchase>30% ∧ sold>100                  → "go_repurchase"
      Rule 4:  sold>1000 ∧ wanted>100                      → "go_hot_wanted"
      Rule 5:  sold>1000 ∧ repurchase>10%                  → "go_hot_repeat"
      Rule 6:  sold>1000 ∧ repurchase=None ∧ wanted≤100    → "trial_hot_unknown"
      Rule 7:  sold>1000 ∧ repurchase≤10%                  → "caution_hot_low"
      Rule 8:  wanted>100 ∧ sold≤100                       → "trial_wanted_low"
      Rule 9:  repurchase>30% ∧ sold≤100                   → "caution_rep_low"
      Rule 10: medium_highlights≥3                          → "trial_medium3"
      Rule 11: medium_highlights 1-2                        → "watch_medium12"
      Rule 12: 其余                                         → "watch_flat"
    """

def _product_verdict(tier, signals) -> dict[str, Any]
    """tier → verdict dict。组装 {score, max_score, grade, tier, summary, verdict, dimensions}"""

# ---- 供应商评判 ----
def _supplier_signals(mapped) -> dict[str, Any]
    """提取 3 维信号强度（强/中等/弱/未知）。
    {identity_strength, cert_strength, years_strength,
     identity_label, cert_label, years_label,
     has_id, has_cert, has_years}
    """

def _supplier_fatal(signals) -> dict[str, Any] | None
    """黑箱检查：身份+认证+年限三项全空 → 致命"""

def _supplier_missing(signals) -> bool
    """数据不足：身份+年限 ≥2 项缺失 → True"""

def _supplier_tier(signals) -> str
    """信号强度计数 → tier key。
      Rule 3: 强信号≥2         → "trust_strong2"
      Rule 4: 强=1 ∨ 中等≥2   → "usable_ok"
      Rule 5: 弱≥2            → "caution_weak2"
      Rule 6: 其余             → "caution_weak2"
    """

def _supplier_verdict(tier, signals) -> dict[str, Any]
    """tier → verdict dict。组装 {score, max_score, grade, tier, summary, verdict, dimensions}"""
```

### 3.4 三个公开函数重写

`evaluate_product()` / `evaluate_supplier()` / `evaluate_summary()` 完全替换，新结构：

```python
def evaluate_product(mapped):
    signals = _product_signals(mapped)
    fatal = _product_fatal(signals)
    if fatal:
        return fatal
    if _product_missing(signals):
        return _product_verdict("skip", signals)
    tier = _product_tier(signals)
    return _product_verdict(tier, signals)

def evaluate_supplier(mapped):
    signals = _supplier_signals(mapped)
    fatal = _supplier_fatal(signals)
    if fatal:
        return fatal
    if _supplier_missing(signals):
        return _supplier_verdict("skip", signals)
    tier = _supplier_tier(signals)
    return _supplier_verdict(tier, signals)

def evaluate_summary(product, supplier):
    p_tier = product.get("tier", "watch")
    s_tier = supplier.get("tier", "caution_weak2")
    summary_tier = SUMMARY_MATRIX[p_tier][s_tier]
    return _summary_verdict(summary_tier, product, supplier)
```

### 3.5 返回值结构变化（关键）

**旧返回：**
```python
{"score": 12, "max_score": 15, "grade": "go", "summary": _verdict(...), "verdict": _verdict(...),
 "dimensions": [5 dims]}
```

**新返回：**
```python
{"score": 12, "max_score": 18, "grade": "go", "tier": "go_hot_wanted",
 "summary": _verdict("prod_summary_go"), "verdict": _verdict("prod_verdict_go_hot_wanted", ...),
 "dimensions": [6 dims],
 "fatal_reason": None,   # 仅 fatal 时有值
 "skip_reason": None,    # 仅 skip 时有值
 "signals": {...}}        # 原始信号快照，供 display_builder 提取注意事项
```

**display_builder 依赖的字段：** `score`, `max_score`, `grade`, `summary`, `verdict`, `dimensions` — 全部保留，结构不变。`tier` 和 `signals` 为新增。

### 3.6 删除内容

- `_score_grade()` — 不再需要比例判定
- `_select_supplier_human()` — 被信号强度引擎替代
- 旧 D1-D5 维度构建代码块 — 被 `_product_signals()` 替代

### 3.7 风险点

- **display_builder.py L420-421 硬编码 `max_score: 15`** — 需改为从 evaluator 返回值取，或代码里同步改为 18（产品 6 维×3=18）
- **display_builder.py L528 `max_score: 9`** — 供应商不变
- **inspect.js L588 `pe.max_score \|\| 15`** — fallback 值 15 需改为 18（产品），或不改（后端总是返回实际值）
